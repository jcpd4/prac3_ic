cmake_minimum_required(VERSION 3.15)
project(knn2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(cv main.cc crossValidation.cc mnist.cc knn.cc sample.cc)
target_include_directories(cv PRIVATE ${PROJECT_SOURCE_DIR})
# Practica 3: secuencial parametrizado y version MPI
add_executable(knn_seq main_seq.cc crossValidation.cc mnist.cc knn.cc sample.cc)
target_include_directories(knn_seq PRIVATE ${PROJECT_SOURCE_DIR})
find_package(MPI COMPONENTS CXX)
if(MPI_CXX_FOUND)
  add_executable(knn_mpi knn_mpi.cc mnist.cc sample.cc)
  target_include_directories(knn_mpi PRIVATE ${PROJECT_SOURCE_DIR})
  target_link_libraries(knn_mpi PRIVATE MPI::MPI_CXX)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(knn2_tests tests.cc sample.cc knn.cc mnist.cc crossValidation.cc)
  add_test(NAME knn2_tests COMMAND knn2_tests)
endif()
